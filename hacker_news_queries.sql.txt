-- 1. Top 5 Stories

SELECT title, score
FROM hacker_news
ORDER BY score DESC
LIMIT 5;


-- 2. Total Score Across All Stories

SELECT SUM(score) AS total_score
FROM hacker_news;


-- 3. Users With More Than 200 Combined Points

SELECT
    user,
    SUM(score) AS total_score
FROM hacker_news
GROUP BY user
HAVING SUM(score) > 200
ORDER BY total_score DESC;


-- 4. Percentage Contribution of Top Users

SELECT (517 + 309 + 304 + 282) / 6366.0 AS fraction;


-- 5. Rickroll Link Analysis

SELECT
    user,
    COUNT(*) AS links_posted
FROM hacker_news
WHERE url LIKE 'https://www.youtube.com/watch?v=dQw4w9WgXcQ'
GROUP BY user;


-- 6. Categorizing Story Sources

SELECT
    CASE
        WHEN url LIKE '%github.com%' THEN 'GitHub'
        WHEN url LIKE '%medium.com%' THEN 'Medium'
        WHEN url LIKE '%nytimes.com%' THEN 'New York Times'
        ELSE 'Other'
    END AS source,
    COUNT(*) AS story_count
FROM hacker_news
GROUP BY source;


-- 7. View Timestamp Format

SELECT timestamp
FROM hacker_news
LIMIT 10;


-- 8. Extract Hour From Timestamp

SELECT
    timestamp,
    STRFTIME('%H', timestamp)
FROM hacker_news
GROUP BY 1
LIMIT 20;


-- 9. Average Score By Hour

SELECT
    STRFTIME('%H', timestamp) AS hour,
    AVG(score) AS average_score
FROM hacker_news
GROUP BY hour
ORDER BY average_score DESC;


-- 10. Final Posting Time Analysis

SELECT
    STRFTIME('%H', timestamp) AS hour,
    ROUND(AVG(score), 2) AS average_score,
    COUNT(*) AS story_count
FROM hacker_news
WHERE timestamp IS NOT NULL
GROUP BY hour
ORDER BY average_score DESC;
